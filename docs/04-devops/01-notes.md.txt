### Контейнеры, Docker
- запуск докер контейнер с Postgre и контроль работы. Ссылка на docker image - https://hub.docker.com/_/postgres
    docker run --name some-postgres -e POSTGRES_PASSWORD=mysecretpassword -d -p 5432:5432 postgres:15.4
    docker exec -it some-postgres bash
    docker stop `docker ps  | grep postgres | awk '{ print $1}'`
    docker rm `docker ps -a | grep postgres | awk '{ print $1}'`
    sudo ss -tln

- сборка контейнера приложения
    export VERSION=0.0.1
    docker build -t app-spring:$VERSION --build-arg VERSION=$VERSION -f app.Dockerfile ../../
    docker run  -p 8080:8080 app-sprint:$VERSION

- сборка через docker compose
    docker compose -f docker-compose-minio.yml up -d
    docker rm `docker ps -a | grep -v CONTAINER | awk ' { a = a " "  $1 } END { print a }' `
    docker stop `docker ps -a | grep -v CONTAINER | awk ' { a = a " "  $1 } END { print a }' `
    docker rmi `docker images | grep -v CONTAINER | grep none | awk ' { a = a " "  $3 } END { print a }' `

- просмотр логов nginx
    docker logs `docker ps | grep nginx | awk '{ print $1}'`

- миграция
    export VERSION=0.0.1
    docker build -t migration:$VERSION --build-arg VERSION=$VERSION -f liqbase.Dockerfile ../../
    set -a; source .env; set +a
    docker run -e LIQUIBASE_COMMAND_URL=$DB_URL -e LIQUIBASE_COMMAND_USERNAME=$DB_USER -e LIQUIBASE_COMMAND_PASSWORD=$DB_PASS migration:$VERSION
